# CUDA is optional: the existing Ogre2 / Python backends stay buildable without it.
option(AGV_ENABLE_CUDA "Build the batched CUDA grid prototype" ON)
if(AGV_ENABLE_CUDA)
  include(CheckLanguage)
  if(NOT CMAKE_CUDA_COMPILER AND EXISTS "/usr/local/cuda/bin/nvcc")
    set(CMAKE_CUDA_COMPILER "/usr/local/cuda/bin/nvcc" CACHE FILEPATH "CUDA compiler")
  endif()
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architectures; includes PTX for newer GPUs")
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(agv_cuda_grid STATIC src/cuda_grid.cu src/cuda_tiles.cu)
    set_target_properties(agv_cuda_grid PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17)
    target_include_directories(agv_cuda_grid PUBLIC include)
    target_link_libraries(agv_cuda_grid PUBLIC CUDA::cudart nlohmann_json::nlohmann_json)
    target_compile_definitions(agv_gz_linescan PRIVATE AGV_HAS_CUDA)
    target_link_libraries(agv_gz_linescan agv_cuda_grid)
    set(AGV_OPTIX_SDK "$ENV{HOME}/opt/optix-sdk-9.1.0" CACHE PATH "Optional OptiX SDK")
    option(AGV_ENABLE_OPTIX "Build experimental OptiX line-scan backend" ON)
    if(AGV_ENABLE_OPTIX AND EXISTS "${AGV_OPTIX_SDK}/include/optix.h")
      find_package(OpenSSL REQUIRED)
      set(OPTIX_PTX "${CMAKE_CURRENT_BINARY_DIR}/agv_optix_scan.ptx")
      add_custom_command(OUTPUT "${OPTIX_PTX}"
        COMMAND ${CMAKE_CUDA_COMPILER} --ptx -O3 -std=c++17 -arch=compute_75
          -I${AGV_OPTIX_SDK}/include -I${CMAKE_CURRENT_SOURCE_DIR}/include
          ${CMAKE_CURRENT_SOURCE_DIR}/src/optix/scan.cu -o "${OPTIX_PTX}"
        DEPENDS src/optix/scan.cu src/optix/shared.h include/agv_linescan/optix_scene.hpp
          include/agv_linescan/cuda_grid.hpp include/agv_linescan/radiometry.hpp)
      add_custom_target(agv_optix_ptx ALL DEPENDS "${OPTIX_PTX}")
      add_library(agv_runtime_material STATIC src/optix/runtime_material.cu)
      set_target_properties(agv_runtime_material PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17)
      target_compile_options(agv_runtime_material PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
      target_link_libraries(agv_runtime_material PUBLIC CUDA::cudart nlohmann_json::nlohmann_json OpenSSL::Crypto)
      add_library(agv_optix_scene STATIC src/optix/scene.cpp src/optix/material_cache.cpp)
      set_target_properties(agv_optix_scene PROPERTIES POSITION_INDEPENDENT_CODE ON)
      target_include_directories(agv_optix_scene PUBLIC include PRIVATE ${AGV_OPTIX_SDK}/include)
      target_link_libraries(agv_optix_scene PUBLIC agv_runtime_material CUDA::cudart nlohmann_json::nlohmann_json OpenSSL::Crypto ${CMAKE_DL_LIBS})
      add_dependencies(agv_optix_scene agv_optix_ptx)
      add_executable(benchmark_led_convergence tools/benchmark_led_convergence.cpp)
      target_link_libraries(benchmark_led_convergence agv_optix_scene yaml-cpp)
      install(TARGETS benchmark_led_convergence DESTINATION lib/${PROJECT_NAME})
      add_executable(benchmark_grooves tools/benchmark_grooves.cpp)
      target_link_libraries(benchmark_grooves agv_optix_scene yaml-cpp)
      install(TARGETS benchmark_grooves DESTINATION lib/${PROJECT_NAME})
      add_executable(benchmark_optix_stream tools/benchmark_optix_stream.cpp)
      target_link_libraries(benchmark_optix_stream agv_optix_scene yaml-cpp gz-sim8::gz-sim8)
      ament_target_dependencies(benchmark_optix_stream rclcpp sensor_msgs std_msgs)
      install(TARGETS benchmark_optix_stream DESTINATION lib/${PROJECT_NAME})
      target_link_libraries(agv_gz_linescan agv_optix_scene)
      target_compile_definitions(agv_gz_linescan PRIVATE AGV_HAS_OPTIX)
      install(FILES "${OPTIX_PTX}" DESTINATION share/${PROJECT_NAME})
    endif()
    add_executable(benchmark_cuda_grid tools/benchmark_cuda_grid.cpp)
    target_link_libraries(benchmark_cuda_grid agv_cuda_grid yaml-cpp nlohmann_json::nlohmann_json gz-sim8::gz-sim8)
    ament_target_dependencies(benchmark_cuda_grid rclcpp sensor_msgs std_msgs)
    install(TARGETS benchmark_cuda_grid DESTINATION lib/${PROJECT_NAME})
  else()
    message(STATUS "CUDA not available; cuda_grid backend disabled")
  endif()
endif()
