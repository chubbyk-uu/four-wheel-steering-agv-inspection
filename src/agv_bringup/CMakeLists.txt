cmake_minimum_required(VERSION 3.16)
project(agv_bringup)
# Keep explicit Debug/multi-config builds available; normal simulation needs optimization.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch worlds config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/follow_camera.py scripts/visualization_tf.py DESTINATION lib/${PROJECT_NAME})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTORCC ON)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz_rendering_vendor REQUIRED)
find_package(gz-rendering8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Quick)
add_library(FollowCameraZoom SHARED gui/FollowCameraZoom.cpp gui/FollowCameraZoom.qrc)
target_link_libraries(FollowCameraZoom PRIVATE gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register gz-transport13::gz-transport13 Qt5::Quick)
install(TARGETS FollowCameraZoom LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_gui_startup test/test_gui_startup.py)
endif()
ament_package()
