cmake_minimum_required(VERSION 3.16)
project(agv_rviz)
# Keep explicit Debug/multi-config builds available; normal simulation needs optimization.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
add_library(agv_mission_panel SHARED src/mission_panel.cpp)
ament_target_dependencies(agv_mission_panel rviz_common pluginlib rclcpp std_msgs)
target_link_libraries(agv_mission_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS agv_mission_panel LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_panel test/test_panel.cpp)
  add_dependencies(test_panel agv_mission_panel)
  target_include_directories(test_panel PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/agv_mission_panel_autogen/include)
  ament_target_dependencies(test_panel rviz_common pluginlib rclcpp std_msgs)
  target_link_libraries(test_panel Qt5::Widgets Qt5::Test)
  add_test(NAME test_panel COMMAND test_panel -platform offscreen)
endif()
ament_package()
