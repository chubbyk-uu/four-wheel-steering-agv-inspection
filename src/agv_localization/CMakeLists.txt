cmake_minimum_required(VERSION 3.16)
project(agv_localization)
# Keep explicit Debug/multi-config builds available; normal simulation needs optimization.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/gnss_simulator scripts/measurement_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_measurements test/test_measurements.py)
  ament_add_pytest_test(test_localization_offload test/test_localization_offload.py)
endif()
ament_package()
